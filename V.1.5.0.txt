<?php
session_start();
header('Content-Type: application/json; charset=utf-8');

// مسیر فایل‌های داده
define('DATA_FILE', './data/data.json');
define('USERS_FILE', './data/users.json');
define('ISSUES_FILE', './data/issues.json');

// --- بررسی احراز هویت ---
if (!isset($_SESSION['user_id'])) {
    http_response_code(401);
    echo json_encode(['success' => false, 'message' => 'Unauthorized']);
    exit;
}

// --- اطلاعات کاربر فعلی از سشن ---
$CURRENT_USER_ID = $_SESSION['user_id'];
$CURRENT_USER_ROLE = $_SESSION['role'];
$CURRENT_USER_DEPT_ID = $_SESSION['departmentId'];
$CURRENT_USER_PERMS = $_SESSION['permissions'];

// --- توابع کمکی خواندن/نوشتن ---
function readData($file) {
    if (!file_exists($file)) return null;
    $content = file_get_contents($file);
    if ($content === false) return null;
    return json_decode($content, true);
}
function writeData($file, $data) {
    return file_put_contents($file, json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
}

// --- مدیریت اکشن‌ها ---
$action = $_GET['action'] ?? '';
$input = json_decode(file_get_contents('php://input'), true);

try {
    if (readData(DATA_FILE) === null || readData(USERS_FILE) === null || readData(ISSUES_FILE) === null) {
        throw new Exception('خطای بحرانی: یک یا چند فایل داده (data.json, users.json, issues.json) یافت نشدند یا قابل خواندن نیستند.');
    }

    switch ($action) {
        
        case 'getData':
            $data = readData(DATA_FILE);
            $all_users = readData(USERS_FILE);
            $filteredEvents = [];

            if ($CURRENT_USER_ROLE === 'manager' || $CURRENT_USER_ROLE === 'evaluator') {
                foreach ($data['events'] as $event) {
                    if ($event['status'] === 'approved' || $event['status'] === 'pending') {
                        $filteredEvents[] = $event;
                    }
                }
            } else { 
                $view_all_depts = $CURRENT_USER_PERMS['view_all_depts'] ?? false;
                $view_depts = $CURRENT_USER_PERMS['view_depts'] ?? [];
                $view_all_users = $CURRENT_USER_PERMS['view_all_users'] ?? false;
                $view_users = $CURRENT_USER_PERMS['view_users'] ?? [];

                foreach ($data['events'] as $event) {
                    $isOwnEvent = $event['proposerId'] === $CURRENT_USER_ID;

                    if ($isOwnEvent) {
                        $filteredEvents[] = $event;
                        continue;
                    }
                    if ($event['status'] !== 'approved') {
                        continue;
                    }
                    
                    $hasDeptAccess = $view_all_depts || in_array($event['departmentId'], $view_depts);
                    $hasUserAccess = $view_all_users || in_array($event['proposerId'], $view_users);
                    
                    if ($hasDeptAccess || $hasUserAccess) {
                        $filteredEvents[] = $event;
                    }
                }
            }
            
            $data['events'] = $filteredEvents;
            $data['users'] = array_map(function($user) {
                unset($user['password_hash']);
                return $user;
            }, $all_users);

            echo json_encode(['success' => true, 'data' => $data]);
            break;

        case 'saveSettings':
            // این اکشن اکنون هم توسط مدیر (در مودال) و هم توسط همه (برای پیمایش هفته) استفاده می‌شود
            // if ($CURRENT_USER_ROLE !== 'manager') throw new Exception('Access Denied'); // این محدودیت را برمیداریم
            $data = readData(DATA_FILE);
            if (isset($input['headerText'])) {
                 $data['settings']['headerText'] = $input['headerText'];
            }
            if (isset($input['weekStartDate'])) {
                 $data['settings']['weekStartDate'] = $input['weekStartDate'];
            }
            writeData(DATA_FILE, $data);
            echo json_encode(['success' => true, 'message' => 'تنظیمات ذخیره شد']);
            break;

        // --- (اصلاح‌شده) ذخیره رویداد ---
        case 'saveEvent':
            if ($CURRENT_USER_ROLE === 'viewer') throw new Exception('Access Denied');
            
            $data = readData(DATA_FILE);
            $eventData = $input['event'];
            $isEditing = !empty($eventData['id']);

            // (جدید) مدیریت رویداد تمام روز
            if ($eventData['isAllDay']) {
                $eventData['startTime'] = "00:00";
                // اگر تمام روز بود، مطمئن شو پایان روز است
                if ($eventData['endDay'] < $eventData['startDay']) {
                    $eventData['endDay'] = $eventData['startDay'];
                }
                $eventData['endTime'] = "24:00";
            }

            // اعتبار سنجی روز/زمان
            if ($eventData['endDay'] < $eventData['startDay']) {
                throw new Exception("روز پایان نمی‌تواند قبل از روز شروع باشد.");
            }
            if ($eventData['endDay'] == $eventData['startDay'] && $eventData['endTime'] <= $eventData['startTime']) {
                throw new Exception("ساعت پایان باید بعد از ساعت شروع باشد.");
            }
            
            if ($isEditing) {
                $eventIndex = array_search($eventData['id'], array_column($data['events'], 'id'));
                if ($eventIndex === false) throw new Exception('Event not found');
                $existingEvent = $data['events'][$eventIndex];

                $canEdit = $CURRENT_USER_ROLE === 'manager' || 
                           $CURRENT_USER_ROLE === 'evaluator' ||
                           ($existingEvent['proposerId'] === $CURRENT_USER_ID && $existingEvent['status'] !== 'approved');
                if (!$canEdit) throw new Exception('Access Denied to edit this event');

                $updatedEvent = array_merge($existingEvent, $eventData);
                
                if ($CURRENT_USER_ROLE !== 'manager' && $updatedEvent['status'] === 'rejected') {
                    $updatedEvent['status'] = 'pending';
                }
                if (($CURRENT_USER_ROLE === 'manager' || $CURRENT_USER_ROLE === 'evaluator') && !isset($eventData['status'])) {
                     $updatedEvent['status'] = 'approved';
                }
                
                $data['events'][$eventIndex] = $updatedEvent;
                $message = 'رویداد با موفقیت ویرایش شد';

            } else {
                $dept = null;
                if ($CURRENT_USER_DEPT_ID) {
                    $dept = array_values(array_filter($data['departments'], fn($d) => $d['id'] === $CURRENT_USER_DEPT_ID))[0] ?? null;
                }
                
                $newEvent = $eventData;
                $newEvent['id'] = time() . rand(100, 999);
                $newEvent['proposerId'] = $CURRENT_USER_ID;
                $newEvent['departmentId'] = $CURRENT_USER_DEPT_ID;
                
                if ($CURRENT_USER_ROLE === 'manager' || $CURRENT_USER_ROLE === 'evaluator') {
                    $newEvent['status'] = 'approved';
                    $message = 'رویداد با موفقیت افزوده شد';
                } else { // Proposer
                    $newEvent['status'] = 'pending';
                    $message = 'پیشنهاد شما ثبت شد و در انتظار تایید است';
                    $newEvent['color'] = $dept['color'] ?? '#c8c8c8';
                    $newEvent['textColor'] = $eventData['textColor'] ?? '#ffffff';
                }
                $data['events'][] = $newEvent;
            }
            
            writeData(DATA_FILE, $data);
            echo json_encode(['success' => true, 'message' => $message]);
            break;

        case 'updateEventStatus':
            if ($CURRENT_USER_ROLE !== 'manager' && $CURRENT_USER_ROLE !== 'evaluator') throw new Exception('Access Denied');
            $data = readData(DATA_FILE);
            $eventId = $input['id'];
            $newStatus = $input['status'];
            $eventIndex = array_search($eventId, array_column($data['events'], 'id'));
            if ($eventIndex === false) throw new Exception('Event not found');
            $data['events'][$eventIndex]['status'] = $newStatus;
            writeData(DATA_FILE, $data);
            echo json_encode(['success' => true, 'message' => 'وضعیت رویداد به‌روز شد']);
            break;

        case 'deleteEvent':
            $data = readData(DATA_FILE);
            $eventId = $input['id'];
            $eventIndex = array_search($eventId, array_column($data['events'], 'id'));
            if ($eventIndex === false) throw new Exception('Event not found');
            $existingEvent = $data['events'][$eventIndex];
            $canDelete = $CURRENT_USER_ROLE === 'manager' || 
                         $CURRENT_USER_ROLE === 'evaluator' ||
                         ($existingEvent['proposerId'] === $CURRENT_USER_ID && $existingEvent['status'] !== 'approved');
            if (!$canDelete) throw new Exception('Access Denied');
            array_splice($data['events'], $eventIndex, 1);
            writeData(DATA_FILE, $data);
            echo json_encode(['success' => true, 'message' => 'رویداد حذف شد']);
            break;
            
        case 'manageDepartments':
            if ($CURRENT_USER_ROLE !== 'manager') throw new Exception('Access Denied');
            $data = readData(DATA_FILE);
            $data['departments'] = $input['departments'];
            writeData(DATA_FILE, $data);
            echo json_encode(['success' => true, 'message' => 'دپارتمان‌ها به‌روزرسانی شدند']);
            break;

        case 'manageUsers':
            if ($CURRENT_USER_ROLE !== 'manager') throw new Exception('Access Denied');
            $users = readData(USERS_FILE);
            $sub_action = $input['sub_action'] ?? '';
            $clientUser = $input['user'] ?? null;
            switch ($sub_action) {
                case 'delete':
                    if (empty($clientUser['id'])) throw new Exception('User ID required');
                    if ($clientUser['id'] === $CURRENT_USER_ID) throw new Exception('شما نمی‌توانید حساب کاربری خود را حذف کنید.');
                    $users = array_filter($users, fn($u) => $u['id'] !== $clientUser['id']);
                    $message = 'کاربر با موفقیت حذف شد';
                    break;
                case 'save':
                    if (empty($clientUser['username'])) throw new Exception('نام کاربری الزامی است');
                    if (empty($clientUser['displayName'])) throw new Exception('نام نمایشی الزامی است');
                    $isEditing = !empty($clientUser['id']);
                    if ($isEditing) {
                        $userIndex = array_search($clientUser['id'], array_column($users, 'id'));
                        if ($userIndex === false) throw new Exception('User not found');
                        $users[$userIndex]['username'] = $clientUser['username'];
                        $users[$userIndex]['displayName'] = $clientUser['displayName'];
                        $users[$userIndex]['role'] = $clientUser['role'];
                        $users[$userIndex]['departmentId'] = $clientUser['departmentId'];
                        $users[$userIndex]['permissions'] = $clientUser['permissions'];
                        if (!empty($clientUser['password'])) {
                            $users[$userIndex]['password_hash'] = password_hash($clientUser['password'], PASSWORD_DEFAULT);
                        }
                        $message = 'کاربر با موفقیت ویرایش شد';
                    } else {
                        if (empty($clientUser['password'])) throw new Exception('رمز عبور برای کاربر جدید الزامی است');
                        $newUser = $clientUser;
                        $newUser['id'] = 'u_' . time();
                        $newUser['password_hash'] = password_hash($clientUser['password'], PASSWORD_DEFAULT);
                        unset($newUser['password']);
                        $users[] = $newUser;
                        $message = 'کاربر با موفقیت افزوده شد';
                    }
                    break;
                default:
                    throw new Exception('Invalid sub-action for manageUsers');
            }
            writeData(USERS_FILE, array_values($users));
            $updatedUsers = array_map(function($user) {
                unset($user['password_hash']);
                return $user;
            }, readData(USERS_FILE));
            echo json_encode(['success' => true, 'message' => $message, 'data' => $updatedUsers]);
            break;

        case 'reportIssue':
            $issues = readData(ISSUES_FILE);
            $newIssue = [
                'id' => time(),
                'userId' => $CURRENT_USER_ID,
                'userName' => $_SESSION['displayName'],
                'text' => $input['text'],
                'timestamp' => date('c'),
                'status' => 'new'
            ];
            $issues[] = $newIssue;
            writeData(ISSUES_FILE, $issues);
            echo json_encode(['success' => true, 'message' => 'پیام شما با موفقیت ثبت شد.']);
            break;
            
        case 'getIssues':
            if ($CURRENT_USER_ROLE !== 'manager') throw new Exception('Access Denied');
            $issues = readData(ISSUES_FILE);
            usort($issues, fn($a, $b) => $b['timestamp'] <=> $a['timestamp']);
            echo json_encode(['success' => true, 'data' => $issues]);
            break;

        default:
            http_response_code(400);
            throw new Exception('Invalid Action');
    }
} catch (Exception $e) {
    http_response_code(500);
    echo json_encode(['success' => false, 'message' => $e->getMessage(), 'trace' => $e->getTraceAsString()]);
}

    // ===================================================================

    async function apiCall(action, body = null, method = 'POST') {
        showLoader();
        try {
            const options = {
                method: method,
                headers: { 'Content-Type': 'application/json' },
            };
            const url = `api.php?action=${action}`;
            if (body) {
                options.method = 'POST';
                options.body = JSON.stringify(body);
            }
            const response = await fetch(url, options);
            const responseText = await response.text();
            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}. Response: ${responseText}`);
            }
            const result = JSON.parse(responseText);
            if (!result.success) {
                console.error('API Error Payload:', result);
                throw new Error(result.message || 'خطای ناشناخته از سرور');
            }
            return result;
        } catch (error) {
            if (error instanceof SyntaxError) {
                console.error("Server returned non-JSON. This is likely a PHP error. Check Network tab > Response.");
                alert("خطا: پاسخ نامعتبر از سرور دریافت شد. (احتمالاً خطای PHP). جزئیات را در کنسول F12 ببینید.");
            } else {
                console.error('API Call Error:', error);
                alert(`خطا در ارتباط با سرور: ${error.message}`);
            }
            return null;
        } finally {
            hideLoader();
        }
    }

    function updateCalendarTitle() {
        const start = new Date(currentWeekStartDate.getTime());
        const end = new Date(start.getTime());
        end.setDate(start.getDate() + 6);

        const formatter = new Intl.DateTimeFormat('fa-IR-u-nu-latn', { day: 'numeric', month: 'long', year: 'numeric' });
        
        // قالب‌بندی بدون سال اگر سال‌ها یکی باشند
        if (start.getFullYear() === end.getFullYear()) {
             const startStr = new Intl.DateTimeFormat('fa-IR-u-nu-latn', { day: 'numeric', month: 'long' }).format(start);
             const endStr = formatter.format(end);
             calendarTitle.textContent = `(${startStr} - ${endStr})`;
        } else {
             calendarTitle.textContent = `(${formatter.format(start)} - ${formatter.format(end)})`;
        }
    }

    // (اصلاح‌شده) منطق رندر برای پشتیبانی از چند روزه
    function renderAllEvents(events) {
        let allDayEventsByDay = Array(7).fill(null).map(() => []);
        let timedEventsByDay = Array(7).fill(null).map(() => []);
        
        // (جدید) پردازش و تکه‌تکه کردن رویدادها
        for (const event of events) {
            for (let day = event.startDay; day <= event.endDay; day++) {
                if (day < 0 || day > 6) continue; // خارج از هفته فعلی

                const chunk = { ...event, day: day }; // کپی کردن رویداد
                
                // تنظیمات چند روزه
                chunk.isMultiDayPart = (event.startDay !== event.endDay);
                chunk.isStart = (day === event.startDay);
                chunk.isEnd = (day === event.endDay);
                
                // تنظیم زمان‌های شروع و پایان برای این "تکه"
                chunk.from = (chunk.isStart) ? event.startTime : "00:00";
                chunk.to = (chunk.isEnd) ? event.endTime : "24:00";
                
                if(event.isAllDay) {
                    allDayEventsByDay[day].push(chunk);
                } else {
                    timedEventsByDay[day].push(chunk);
                }
            }
        }
        
        // رندر کردن تکه‌ها در روزهای مربوطه
        for (let day = 0; day < 7; day++) {
            const dayContainer = document.querySelector(`[data-day-index="${day}"]`);
            if (!dayContainer) continue;
            
            const allDayCell = dayContainer.querySelector('.all-day-cell');
            const allDayEvents = allDayEventsByDay[day];
            if (allDayEvents.length > 0) {
                allDayCell.style.setProperty('--event-count', allDayEvents.length);
                allDayEvents.forEach(event => createEventDiv(event, allDayCell, true));
            }
            
            const timelineCell = dayContainer.querySelector('.timeline-cell');
            const timedEvents = timedEventsByDay[day].sort((a, b) => timeToMinutes(a.from) - timeToMinutes(b.from));
            if (timedEvents.length > 0) processAndRenderTimedEvents(timedEvents, timelineCell);
        }
    }

    function createEventDiv(event, container, isAllDay = false) {
        const eventDiv = document.createElement('div');
        eventDiv.className = isAllDay ? 'event-allday' : 'event';
        eventDiv.classList.add(`status-${event.status || 'approved'}`);

        if (event.status === 'approved') {
            eventDiv.style.backgroundColor = event.color;
            eventDiv.style.color = event.textColor || 'white';
        }
        
        // (جدید) اضافه کردن دیتا برای استایلینگ
        if (event.isMultiDayPart) {
            eventDiv.dataset.ismultiday = true;
            if (event.isStart) eventDiv.dataset.isstart = true;
            if (event.isEnd) eventDiv.dataset.isend = true;
        }
        
        let isLocked = false;
        if (CURRENT_USER.role === 'proposer' || CURRENT_USER.role === 'viewer') {
            if(event.status === 'approved' && event.proposerId !== CURRENT_USER.id) {
                isLocked = true;
            }
        }
        if (CURRENT_USER.role === 'viewer' && event.proposerId) {
            isLocked = true;
        }
        if(isLocked) eventDiv.classList.add('event-locked');
        
        eventDiv.dataset.event = JSON.stringify(event);
        eventDiv.dataset.locked = isLocked;
        
        container.appendChild(eventDiv);
        return eventDiv;
    }

    // (اصلاح‌شده) منطق رندر برای 24 ساعت
    function processAndRenderTimedEvents(dayEvents, timelineCell) {
        const isMobile = document.body.classList.contains('mobile-view');
        dayEvents.forEach(event => { event.start = timeToMinutes(event.from); event.end = timeToMinutes(event.to); });
        
        let i = 0;
        while (i < dayEvents.length) {
            let overlappingGroup = [dayEvents[i]];
            let maxEnd = dayEvents[i].end;
            for (let j = i + 1; j < dayEvents.length; j++) {
                if (dayEvents[j].start < maxEnd) {
                    overlappingGroup.push(dayEvents[j]);
                    if (dayEvents[j].end > maxEnd) maxEnd = dayEvents[j].end;
                } else break;
            }
            
            const lanes = [];
            overlappingGroup.sort((a,b) => a.start - b.start).forEach(event => {
                let placed = false;
                for (const lane of lanes) { if (event.start >= lane[lane.length - 1].end) { lane.push(event); placed = true; break; } }
                if (!placed) lanes.push([event]);
            });
            
            const totalLanes = lanes.length;
            lanes.forEach((lane, laneIndex) => {
                lane.forEach(event => {
                    const eventWrapper = document.createElement('div'); eventWrapper.className = 'event-wrapper';
                    const totalMinutesInView = 24 * 60; // (جدید) 24 ساعت
                    const startOffsetMinutes = event.start; // (جدید) شروع از 0
                    const durationMinutes = event.end - event.start;
                    
                    if (isMobile) {
                        eventWrapper.style.top = `${(startOffsetMinutes / totalMinutesInView) * 100}%`; eventWrapper.style.height = `${(durationMinutes / totalMinutesInView) * 100}%`;
                        eventWrapper.style.left = `${(laneIndex / totalLanes) * 100}%`; eventWrapper.style.width = `${100 / totalLanes}%`;
                    } else {
                        eventWrapper.style.right = `${(startOffsetMinutes / totalMinutesInView) * 100}%`; eventWrapper.style.width = `${(durationMinutes / totalMinutesInView) * 100}%`;
                        eventWrapper.style.top = `${(laneIndex / totalLanes) * 100}%`; eventWrapper.style.height = `${100 / totalLanes}%`;
                    }
                    createEventDiv(event, eventWrapper, false);
                });
            });
            i += overlappingGroup.length;
        }
    }

    // (اصلاح‌شده) دریافت تاریخ شروع
    function getStartDate() {
        const gregorianDateStr = fullData.settings.weekStartDate;
        if (gregorianDateStr && /^\d{4}-\d{2}-\d{2}$/.test(gregorianDateStr)) {
            const [year, month, day] = gregorianDateStr.split('-').map(Number);
            return new Date(year, month - 1, day);
        }
        // Fallback to THIS week's Saturday
        const today = new Date();
        const firstSaturday = new Date(today.setDate(today.getDate() - (today.getDay() + 1) % 7));
        firstSaturday.setHours(0, 0, 0, 0);
        return firstSaturday;
    }

    function generateWeeks() {
        weekSelector.innerHTML = '';
        const today = new Date();
        const currentYear = today.getFullYear();
        
        for(let year = currentYear - 1; year <= currentYear + 1; year++) {
            const yearStart = new Date(year, 0, 1);
            let firstSaturday = new Date(yearStart);
            firstSaturday.setDate(yearStart.getDate() - (yearStart.getDay() + 1) % 7);

            for (let i = 0; i < 53; i++) {
                const weekStart = new Date(firstSaturday);
                weekStart.setDate(firstSaturday.getDate() + (i * 7));
                if (weekStart.getFullYear() > year) break;
                
                const weekEnd = new Date(weekStart);
                weekEnd.setDate(weekEnd.getDate() + 6);
                
                const option = document.createElement('option');
                const gregorianStartDate = formatDate(weekStart);
                option.value = gregorianStartDate;
                
                const startStr = new Intl.DateTimeFormat('fa-IR', { day: 'numeric', month: 'long' }).format(weekStart);
                const endStr = new Intl.DateTimeFormat('fa-IR', { day: 'numeric', month: 'long' }).format(weekEnd);
                option.textContent = `(${startStr} - ${endStr}, ${toPersian(year)})`; // (جدید) قالب تاریخ
                
                weekSelector.appendChild(option);
            }
        }
    }

import os

# --- تنظیمات ---
# پوشه‌ای که می‌خواهید اسکن کنید (نقطه '.' به معنای پوشه فعلی است)
ROOT_DIRECTORY = '.' 

# نام فایل نهایی که همه‌چیز در آن ذخیره می‌شود
OUTPUT_FILE = 'text.txt'

# لیست پوشه‌هایی که باید نادیده گرفته شوند
IGNORED_DIRECTORIES = {
    '.git', 
    '.gitignore', 
    '.vscode', 
    'node_modules', 
    '__pycache__', 
    '.venv',
    'venv',
    '__pycache__',
    'dist',
    'build',
    '.next',
    '.node_modules',
    'migrations'
}

# لیست پسوند فایل‌هایی که باید نادیده گرفته شوند (فایل‌های باینری، عکس و...)
IGNORED_EXTENSIONS = {
    '.png', '.jpg', '.jpeg', '.gif', '.bmp', '.ico', '.svg',
    '.zip', '.rar', '.gz', '.tar',
    '.pdf', '.doc', '.docx', '.xls', '.xlsx',
    '.exe', '.dll', '.so', '.o',
    '.pyc',
    '.mp4', '.mkv', '.mov', '.mp3',
    '.lock'
}
# --- پایان تنظیمات ---

def get_all_files_content():
    # مسیر کامل فایل خروجی را می‌سازیم تا بعداً بتوانیم آن را نادیده بگیریم
    try:
        abs_output_path = os.path.abspath(os.path.join(ROOT_DIRECTORY, OUTPUT_FILE))
    except Exception as e:
        print(f"Error getting absolute path for output file: {e}")
        return

    print(f"Writing project content to {OUTPUT_FILE}...")

    # فایل خروجی را در حالت 'write' با انکودینگ 'utf-8' باز می‌کنیم
    try:
        with open(OUTPUT_FILE, 'w', encoding='utf-8') as outfile:
            # os.walk به صورت بازگشتی تمام پوشه‌ها و فایل‌ها را پیدا می‌کند
            # topdown=True به ما اجازه می‌دهد لیست dirnames را ویرایش کنیم
            for dirpath, dirnames, filenames in os.walk(ROOT_DIRECTORY, topdown=True):
                
                # --- بهینه‌سازی: از ورود به پوشه‌های نادیده گرفته شده جلوگیری می‌کنیم ---
                # ما لیست dirnames را "درجا" (in-place) ویرایش می‌کنیم
                # [:]
                dirnames[:] = [d for d in dirnames if d not in IGNORED_DIRECTORIES]

                for filename in filenames:
                    # مسیر کامل فایل فعلی
                    file_path = os.path.join(dirpath, filename)
                    
                    # مسیر کامل برای مقایسه
                    abs_file_path = os.path.abspath(file_path)

                    # --- نادیده گرفتن فایل‌ها ---
                    # 1. اگر فایل فعلی، خود فایل خروجی است، از آن بگذر
                    if abs_file_path == abs_output_path:
                        continue
                        
                    # 2. اگر پسوند فایل در لیست نادیده گرفته شده‌ها بود، از آن بگذر
                    file_ext = os.path.splitext(filename)[1].lower()
                    if file_ext in IGNORED_EXTENSIONS:
                        continue

                    # --- نوشتن در فایل خروجی ---
                    try:
                        # مسیر نسبی فایل (برای نمایش در خروجی)
                        relative_path = os.path.relpath(file_path, ROOT_DIRECTORY)
                        # اطمینان از اینکه از اسلش / استفاده می‌شود (برای سازگاری)
                        relative_path = relative_path.replace(os.path.sep, '/')

                        # 1. نوشتن مسیر فایل طبق فرمت درخواستی شما
                        outfile.write(f"{relative_path}\n\n")

                        # 2. خواندن محتوای فایل
                        with open(file_path, 'r', encoding='utf-8') as infile:
                            content = infile.read()
                        
                        # 3. نوشتن محتوای فایل
                        outfile.write(content)
                        
                        # 4. اضافه کردن دو خط جدید برای جدا کردن از فایل بعدی
                        outfile.write("\n\n")

                    except UnicodeDecodeError:
                        # این اتفاق زمانی می‌افتد که فایل متنی نیست (مثلاً باینری است)
                        outfile.write(f"[Error: Cannot read binary file or file with unsupported encoding]\n\n")
                    except Exception as e:
                        outfile.write(f"[Error reading file {file_path}: {e}]\n\n")

    except IOError as e:
        print(f"Error opening or writing to output file {OUTPUT_FILE}: {e}")
    except Exception as e:
        print(f"An unexpected error occurred: {e}")
    
    print(f"Done! All content written to {OUTPUT_FILE}")

# اجرای تابع اصلی

    get_all_files_content()
