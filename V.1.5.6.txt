/**
 * Reads JSON data from a file with file locking.
 * Ensures 'events', 'departments', 'holidays' are arrays.
 *  - Added error logging for JSON decode errors.
 *
 * @param string $file The path to the JSON file.
 * @return array|null The decoded data or null on failure.
 */
function readData($file) {
    if (!file_exists($file)) {
        error_log("File not found: $file");
        return null;
    }

    $fp = fopen($file, 'r');
    if (!$fp) {
        error_log("Could not open file for reading: $file");
        return null;
    }

    // Acquire shared lock for reading
    if (!flock($fp, LOCK_SH)) {
        fclose($fp);
        error_log("Could not acquire shared lock for reading: $file");
        return null;
    }

    $content = stream_get_contents($fp);
    flock($fp, LOCK_UN); // Release lock
    fclose($fp);

    if ($content === false) {
        error_log("Could not read file content: $file");
        return null;
    }
    // Handle empty file case
    if (trim($content) === '') {
        if (basename($file) === 'data.json') return ['settings' => [], 'events' => [], 'holidays' => [], 'departments' => []];
        if (basename($file) === 'users.json') return [];
        if (basename($file) === 'issues.json') return [];
        return null;
    }

    $data = json_decode($content, true);

    // Check for JSON decoding errors
    if (json_last_error() !== JSON_ERROR_NONE) {
        error_log("JSON Decode Error in $file: " . json_last_error_msg() . " | Content prefix: " . substr($content, 0, 100));
        // Return a default structure to prevent fatal errors downstream
        if (basename($file) === 'data.json') return ['settings' => [], 'events' => [], 'holidays' => [], 'departments' => []];
        if (basename($file) === 'users.json') return [];
        if (basename($file) === 'issues.json') return [];
        return null; // Or throw exception?
    }

    // Ensure essential keys are arrays (for data.json)
    if (basename($file) === 'data.json') {
        $data['settings'] = isset($data['settings']) && is_array($data['settings']) ? $data['settings'] : [];
        $data['events'] = isset($data['events']) && is_array($data['events']) ? $data['events'] : [];
        $data['departments'] = isset($data['departments']) && is_array($data['departments']) ? $data['departments'] : [];
        $data['holidays'] = isset($data['holidays']) && is_array($data['holidays']) ? $data['holidays'] : [];
    }
    // Ensure root is an array for users and issues
    elseif ((basename($file) === 'users.json' || basename($file) === 'issues.json') && !is_array($data)) {
        error_log("Invalid structure in $file: expected root array.");
        $data = []; // Reset to empty array if structure is wrong
    }


    return $data;
}

/**
 * Writes JSON data to a file with file locking and sorting for events.
 *  - Added Exception handling for date comparison.
 *
 * @param string $file The path to the JSON file.
 * @param mixed $data The data to encode and write.
 * @return bool True on success, false on failure.
 */
function writeData($file, $data) {
    // Ensure events array exists and sort it before writing (only for data.json)
    if (basename($file) === 'data.json') {
        if (!isset($data['events']) || !is_array($data['events'])) {
            $data['events'] = [];
        }
        usort($data['events'], function ($a, $b) {
            $startA = $a['startDateTime'] ?? '9999-12-31T23:59:59Z'; // Use Z for UTC comparison if no offset
            $startB = $b['startDateTime'] ?? '9999-12-31T23:59:59Z';
            try {
                // DateTime constructor correctly handles ISO strings with or without offsets
                $dtA = new DateTime($startA);
                $dtB = new DateTime($startB);
                return $dtA <=> $dtB; // Use spaceship operator for comparison
            } catch (Exception $e) {
                 // Log error and treat as equal to avoid breaking sort
                 error_log("Date comparison error during sort: " . $e->getMessage() . " | Comparing: '$startA' <=> '$startB'");
                 return 0;
            }
        });
    }

    $fp = fopen($file, 'c'); // Open for writing, create if not exists, pointer at beginning
    if (!$fp) {
        error_log("Could not open file for writing: $file");
        return false;
    }

    // Acquire exclusive lock for writing
    if (!flock($fp, LOCK_EX)) {
        fclose($fp);
        error_log("Could not acquire exclusive lock for writing: $file");
        return false;
    }

    // Truncate the file to zero length AFTER acquiring the lock
    ftruncate($fp, 0);

    $writeResult = fwrite($fp, json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE | JSON_INVALID_UTF8_SUBSTITUTE)); // Added UTF8 substitute
    fflush($fp); // Ensure data is written to disk
    flock($fp, LOCK_UN); // Release lock
    fclose($fp);

    if ($writeResult === false) {
        error_log("Failed to write data to file: $file");
    }
    return $writeResult !== false;
}

// --- Input Sanitization Helper (Basic) ---
/**
 * Basic string sanitization using strip_tags.
 *  - Added null check.
 *
 * @param string|null $input The string to sanitize.
 * @return string The sanitized string.
 */
function sanitizeString($input) {
    // Check if input is null or not a string, return empty string in those cases
    if ($input === null || !is_string($input)) {
        return '';
    }
    // strip_tags removes HTML and PHP tags. Consider allowing specific tags if needed later.
    return strip_tags($input);
}


// --- Action Handling ---
$action = $_GET['action'] ?? '';
$input = json_decode(file_get_contents('php://input'), true);

try {
    // Read initial data using locked read function
    $appData = readData(DATA_FILE);
    $usersData = readData(USERS_FILE); // Array of users
    $issuesData = readData(ISSUES_FILE); // Array of issues

    // Critical check: Ensure data files were read correctly
    if ($appData === null || $usersData === null || $issuesData === null) {
        throw new Exception('خطای بحرانی: یک یا چند فایل داده اصلی قابل خواندن نیستند.');
    }

    switch ($action) {

        case 'getData':
            $all_users = $usersData; // Users data read securely

            // Determine the target week start date
            $weekStartDateStr = $_GET['weekStartDate'] ?? ($appData['settings']['weekStartDate'] ?? null);
            if (!$weekStartDateStr || !preg_match('/^\d{4}-\d{2}-\d{2}$/', $weekStartDateStr)) {
                // Default to Saturday of the current week (based on server time in Tehran)
                $todayDt = new DateTime('now', new DateTimeZone('Asia/Tehran'));
                $diffToSat = ($todayDt->format('w') + 1) % 7;
                $todayDt->modify("-$diffToSat days");
                $weekStartDateStr = $todayDt->format('Y-m-d');
            }

            // Create DateTime objects for the week range in Tehran time
            $weekStartDT = new DateTime($weekStartDateStr . ' 00:00:00', new DateTimeZone('Asia/Tehran'));
            $weekEndDT = clone $weekStartDT;
            $weekEndDT->modify('+7 days'); // End date is exclusive (start of the next week)

            // --- Filter Events by Date Range ---
            $dateFilteredEvents = [];
            foreach ($appData['events'] as $event) {
                 if (empty($event['startDateTime']) || empty($event['endDateTime']) || !is_string($event['startDateTime']) || !is_string($event['endDateTime'])) {
                     error_log("Skipping event due to missing or non-string DateTime: " . ($event['id'] ?? 'N/A'));
                     continue;
                 }
                 try {
                     $eventStartDT = new DateTime($event['startDateTime']); // Handles offsets
                     $endStr = $event['endDateTime'];
                     $eventEndDT = null;

                     if (str_contains($endStr, 'T24:00:00')) {
                         $datePart = substr($endStr, 0, 10);
                         $offsetPart = substr($endStr, 19) ?: date('P'); // Use current offset if missing
                         $eventEndDT = new DateTime($datePart . 'T00:00:00' . $offsetPart);
                         $eventEndDT->modify('+1 day');
                     } else {
                         $eventEndDT = new DateTime($endStr);
                     }

                    // Check for overlap: event starts before week ends AND event ends after week starts
                    if ($eventStartDT < $weekEndDT && $eventEndDT > $weekStartDT) {
                        $dateFilteredEvents[] = $event;
                    }
                 } catch (Exception $e) {
                     error_log("Skipping event due to invalid DateTime format during getData filtering: ID " . ($event['id'] ?? 'N/A') . " - Error: " . $e->getMessage());
                     continue;
                 }
            }

            // --- Access Control Filtering ---
            $filteredEvents = [];
            if ($CURRENT_USER_ROLE === ROLE_MANAGER || $CURRENT_USER_ROLE === ROLE_EVALUATOR) {
                // Manager/Evaluator see all Approved and Pending events in the date range
                $filteredEvents = array_filter($dateFilteredEvents, function ($event) {
                    $status = $event['status'] ?? STATUS_APPROVED; // Assume approved if status missing? Be careful.
                    return $status === STATUS_APPROVED || $status === STATUS_PENDING;
                });
            } else { // Proposer or Viewer
                 $view_all_depts = $CURRENT_USER_PERMS['view_all_depts'] ?? false;
                 $view_depts = $CURRENT_USER_PERMS['view_depts'] ?? [];
                 $view_all_users = $CURRENT_USER_PERMS['view_all_users'] ?? false;
                 $view_users = $CURRENT_USER_PERMS['view_users'] ?? [];

                 foreach ($dateFilteredEvents as $event) {
                    $proposerId = $event['proposerId'] ?? null;
                    $deptId = $event['departmentId'] ?? null;
                    $status = $event['status'] ?? STATUS_APPROVED;

                    $isOwnEvent = ($proposerId && $proposerId === $CURRENT_USER_ID);
                    if ($isOwnEvent) {
                        $filteredEvents[] = $event; // Show own events regardless of status
                        continue;
                    }
                    if ($status !== STATUS_APPROVED) {
                        continue; // Only show approved events from others
                    }

                    // Check permissions
                    $hasDeptAccess = $view_all_depts || ($deptId && is_array($view_depts) && in_array($deptId, $view_depts));
                    $hasUserAccess = $view_all_users || ($proposerId && is_array($view_users) && in_array($proposerId, $view_users));
                    if ($hasDeptAccess || $hasUserAccess) {
                        $filteredEvents[] = $event;
                    }
                 }
            }
            // Ensure result is a plain array (reset keys after filtering)
            $filteredEvents = array_values($filteredEvents);

            // --- Prepare final output ---
             $outputData = [
                 'settings' => $appData['settings'] ?? [],
                 'events' => $filteredEvents, // Return only filtered events
                 'holidays' => $appData['holidays'] ?? [],
                 'departments' => $appData['departments'] ?? [],
                 'users' => array_map(function($user) { // Remove password hash
                                unset($user['password_hash']);
                                return $user;
                            }, $all_users)
             ];

            echo json_encode(['success' => true, 'data' => $outputData]);
            break;

        case 'saveSettings':
            if ($CURRENT_USER_ROLE !== ROLE_MANAGER) throw new Exception('شما اجازه تغییر تنظیمات را ندارید.');

            if (isset($input['headerText'])) {
                 $appData['settings']['headerText'] = sanitizeString($input['headerText']);
            }
            if (isset($input['weekStartDate']) && preg_match('/^\d{4}-\d{2}-\d{2}$/', $input['weekStartDate'])) {
                 $appData['settings']['weekStartDate'] = $input['weekStartDate'];
            } else {
                 // Optionally log or ignore invalid date format attempt
                 error_log("Attempted to save settings with invalid weekStartDate format: " . ($input['weekStartDate'] ?? 'NULL'));
            }

            if (!writeData(DATA_FILE, $appData)) throw new Exception('خطا در ذخیره تنظیمات.');
            echo json_encode(['success' => true, 'message' => 'تنظیمات با موفقیت ذخیره شد.']);
            break;

        case 'saveEvent':
            if ($CURRENT_USER_ROLE === ROLE_VIEWER) throw new Exception('شما اجازه ذخیره رویداد را ندارید.');

            $eventData = $input['event'] ?? null;
            if (!$eventData) throw new Exception('اطلاعات رویداد ارسال نشده است.');

            $isEditing = !empty($eventData['id']);

            // --- DateTime Validation ---
            if (empty($eventData['startDateTime']) || empty($eventData['endDateTime']) || !is_string($eventData['startDateTime']) || !is_string($eventData['endDateTime'])) {
                 throw new Exception("فرمت تاریخ یا ساعت شروع/پایان نامعتبر است.");
            }
            try {
                 $startDT = new DateTime($eventData['startDateTime']);
                 $endStr = $eventData['endDateTime'];
                 $compareEndDT = null;
                 if (str_contains($endStr, 'T24:00:00')) {
                    $datePart = substr($endStr, 0, 10);
                    $offsetPart = substr($endStr, 19) ?: date('P');
                    $compareEndDT = new DateTime($datePart . 'T00:00:00' . $offsetPart);
                    $compareEndDT->modify('+1 day');
                 } else {
                    $compareEndDT = new DateTime($endStr);
                 }
                if ($compareEndDT <= $startDT) throw new Exception("تاریخ و ساعت پایان باید بعد از شروع باشد.");
            } catch (Exception $e) { throw new Exception("خطا در پردازش تاریخ و ساعت ورودی: " . $e->getMessage()); }
            // --- End DateTime Validation ---

            if ($isEditing) {
                // --- Edit Event ---
                $eventId = $eventData['id'];
                $eventIndex = false;
                foreach ($appData['events'] as $index => $e) { if (isset($e['id']) && (string)$e['id'] === (string)$eventId) { $eventIndex = $index; break; } }
                if ($eventIndex === false) throw new Exception('رویداد مورد نظر یافت نشد.');

                $existingEvent = $appData['events'][$eventIndex];
                if (($existingEvent['isHoliday'] ?? false)) throw new Exception('تعطیلات رسمی قابل ویرایش نیستند.'); // Prevent editing holidays

                // Check Edit Permissions
                $canEdit = ($CURRENT_USER_ROLE === ROLE_MANAGER || $CURRENT_USER_ROLE === ROLE_EVALUATOR ||
                           (($existingEvent['proposerId'] ?? null) === $CURRENT_USER_ID &&
                            (($existingEvent['status'] ?? STATUS_APPROVED) === STATUS_PENDING || ($existingEvent['status'] ?? STATUS_APPROVED) === STATUS_REJECTED)));
                if (!$canEdit) throw new Exception('شما اجازه ویرایش این رویداد را ندارید.');

                // Allowed fields
                 $allowedFields = ['title', 'by', 'for', 'goal', 'isAllDay', 'startDateTime', 'endDateTime'];
                 if ($CURRENT_USER_ROLE === ROLE_MANAGER) $allowedFields = array_merge($allowedFields, ['color', 'textColor', 'status', 'departmentId']);
                 elseif ($CURRENT_USER_ROLE === ROLE_EVALUATOR) $allowedFields[] = 'status';

                 $updatedEvent = $existingEvent;
                 foreach ($allowedFields as $field) {
                     if (array_key_exists($field, $eventData)) { // Use array_key_exists for null values
                         if ($field === 'isAllDay') $updatedEvent[$field] = (bool)$eventData[$field];
                         elseif ($field === 'startDateTime' || $field === 'endDateTime') $updatedEvent[$field] = $eventData[$field]; // Already validated
                         else $updatedEvent[$field] = sanitizeString($eventData[$field]);
                     }
                 }

                // Status Logic: Only apply if allowed and provided, handle resubmission
                 if (in_array('status', $allowedFields) && isset($eventData['status']) && in_array($eventData['status'], [STATUS_APPROVED, STATUS_PENDING, STATUS_REJECTED])) {
                    if ($CURRENT_USER_ROLE === ROLE_PROPOSER && $existingEvent['status'] === STATUS_REJECTED) {
                        $updatedEvent['status'] = STATUS_PENDING; // Resubmitting rejected event makes it pending
                    } else {
                         $updatedEvent['status'] = $eventData['status'];
                    }
                 } // Keep existing status if not changed or not allowed

                $appData['events'][$eventIndex] = $updatedEvent;
                $message = 'رویداد با موفقیت ویرایش شد.';

            } else {
                // --- Add New Event ---
                $deptColor = '#cccccc'; $deptTextColor = '#ffffff';
                if ($CURRENT_USER_DEPT_ID && isset($appData['departments'])) {
                    foreach($appData['departments'] as $dept) {
                        if (($dept['id'] ?? null) === $CURRENT_USER_DEPT_ID) { $deptColor = $dept['color'] ?? $deptColor; break; }
                    }
                }
                $deptTextColor = isLightColor($deptColor) ? '#333333' : '#ffffff';

                $newEvent = [
                    'id' => uniqid('evt_', true),
                    'proposerId' => $CURRENT_USER_ID,
                    'departmentId' => $CURRENT_USER_DEPT_ID,
                    'title' => sanitizeString($eventData['title'] ?? 'رویداد بدون عنوان'),
                    'by' => sanitizeString($eventData['by'] ?? ''),
                    'for' => sanitizeString($eventData['for'] ?? ''),
                    'goal' => sanitizeString($eventData['goal'] ?? ''),
                    'isAllDay' => (bool)($eventData['isAllDay'] ?? false),
                    'startDateTime' => $eventData['startDateTime'],
                    'endDateTime' => $eventData['endDateTime'],
                ];

                if ($CURRENT_USER_ROLE === ROLE_MANAGER || $CURRENT_USER_ROLE === ROLE_EVALUATOR) {
                    $newEvent['status'] = STATUS_APPROVED;
                    $message = 'رویداد با موفقیت افزوده شد.';
                    $newEvent['color'] = ($CURRENT_USER_ROLE === ROLE_MANAGER && isset($eventData['color'])) ? sanitizeString($eventData['color']) : $deptColor;
                    $newEvent['textColor'] = ($CURRENT_USER_ROLE === ROLE_MANAGER && isset($eventData['textColor'])) ? sanitizeString($eventData['textColor']) : (isLightColor($newEvent['color']) ? '#333333' : '#ffffff');
                } else { // Proposer
                    $newEvent['status'] = STATUS_PENDING;
                    $message = 'پیشنهاد شما ثبت شد و در انتظار تایید است.';
                    $newEvent['color'] = $deptColor;
                    $newEvent['textColor'] = $deptTextColor;
                }
                $appData['events'][] = $newEvent;
            }

            if (!writeData(DATA_FILE, $appData)) throw new Exception('خطا در ذخیره رویداد.');
            echo json_encode(['success' => true, 'message' => $message]);
            break;

        case 'updateEventStatus':
            if ($CURRENT_USER_ROLE !== ROLE_MANAGER && $CURRENT_USER_ROLE !== ROLE_EVALUATOR) throw new Exception('شما اجازه تغییر وضعیت رویداد را ندارید.');

            $eventId = sanitizeString($input['id'] ?? null);
            $newStatus = sanitizeString($input['status'] ?? null);
            if (!$eventId || !$newStatus || !in_array($newStatus, [STATUS_APPROVED, STATUS_PENDING, STATUS_REJECTED])) throw new Exception('شناسه رویداد یا وضعیت نامعتبر است.');

            $eventIndex = false;
            foreach ($appData['events'] as $index => $e) { if (isset($e['id']) && (string)$e['id'] === (string)$eventId) { $eventIndex = $index; break; } }
            if ($eventIndex === false) throw new Exception('رویداد مورد نظر یافت نشد.');
            if (($appData['events'][$eventIndex]['isHoliday'] ?? false)) throw new Exception('وضعیت تعطیلات رسمی قابل تغییر نیست.');

            $appData['events'][$eventIndex]['status'] = $newStatus;

            // Apply default colors on approval if none set
            if ($newStatus === STATUS_APPROVED && (!isset($appData['events'][$eventIndex]['color']) || !isset($appData['events'][$eventIndex]['textColor']))) {
                 $proposerDeptId = $appData['events'][$eventIndex]['departmentId'] ?? null;
                 $deptColor = '#5c6bc0'; $deptTextColor = '#ffffff';
                 if ($proposerDeptId && isset($appData['departments'])) {
                     foreach($appData['departments'] as $dept) {
                         if (($dept['id'] ?? null) === $proposerDeptId) { $deptColor = $dept['color'] ?? $deptColor; break; }
                     }
                 }
                 if (!isset($appData['events'][$eventIndex]['color'])) $appData['events'][$eventIndex]['color'] = $deptColor;
                 if (!isset($appData['events'][$eventIndex]['textColor'])) $appData['events'][$eventIndex]['textColor'] = isLightColor($deptColor) ? '#333333' : '#ffffff';
            }

            if (!writeData(DATA_FILE, $appData)) throw new Exception('خطا در به‌روزرسانی وضعیت رویداد.');
            echo json_encode(['success' => true, 'message' => 'وضعیت رویداد با موفقیت به‌روز شد.']);
            break;

        case 'deleteEvent':
            $eventId = sanitizeString($input['id'] ?? null);
            if (!$eventId) throw new Exception('شناسه رویداد نامعتبر است.');

            $eventIndex = false;
            foreach ($appData['events'] as $index => $e) { if (isset($e['id']) && (string)$e['id'] === (string)$eventId) { $eventIndex = $index; break; } }
            if ($eventIndex === false) throw new Exception('رویداد مورد نظر یافت نشد.');
            if (($appData['events'][$eventIndex]['isHoliday'] ?? false)) throw new Exception('تعطیلات رسمی قابل حذف نیستند.');

            $existingEvent = $appData['events'][$eventIndex];
            $canDelete = ($CURRENT_USER_ROLE === ROLE_MANAGER || $CURRENT_USER_ROLE === ROLE_EVALUATOR ||
                         (($existingEvent['proposerId'] ?? null) === $CURRENT_USER_ID && ($existingEvent['status'] ?? '') !== STATUS_APPROVED));
            if (!$canDelete) throw new Exception('شما اجازه حذف این رویداد را ندارید.');

            array_splice($appData['events'], $eventIndex, 1);
            if (!writeData(DATA_FILE, $appData)) throw new Exception('خطا در حذف رویداد.');
            echo json_encode(['success' => true, 'message' => 'رویداد با موفقیت حذف شد.']);
            break;

        case 'manageDepartments':
            if ($CURRENT_USER_ROLE !== ROLE_MANAGER) throw new Exception('شما اجازه مدیریت دپارتمان‌ها را ندارید.');
             $cleanDepartments = [];
             if (isset($input['departments']) && is_array($input['departments'])) {
                 foreach ($input['departments'] as $dept) {
                     $name = sanitizeString($dept['name'] ?? '');
                     $color = sanitizeString($dept['color'] ?? ''); // Basic hex validation could be added
                     $id = sanitizeString($dept['id'] ?? '');
                     if (!empty($name) && !empty($color)) {
                         if (empty($id) || strpos($id, 'new_') === 0) $id = uniqid('dept_', true);
                         $cleanDepartments[] = ['id' => $id, 'name' => $name, 'color' => $color];
                     }
                 }
             }
            $appData['departments'] = $cleanDepartments;
            if (!writeData(DATA_FILE, $appData)) throw new Exception('خطا در ذخیره دپارتمان‌ها.');
            echo json_encode(['success' => true, 'message' => 'دپارتمان‌ها با موفقیت به‌روزرسانی شدند.', 'data' => $cleanDepartments]);
            break;

        case 'manageUsers':
            if ($CURRENT_USER_ROLE !== ROLE_MANAGER) throw new Exception('شما اجازه مدیریت کاربران را ندارید.');
            $sub_action = sanitizeString($input['sub_action'] ?? '');
            $clientUser = $input['user'] ?? null;
            $updatedUsers = $usersData; // Start with current users (array)

            switch ($sub_action) {
                case 'delete':
                    $userIdToDelete = sanitizeString($clientUser['id'] ?? null);
                    if (!$userIdToDelete) throw new Exception('شناسه کاربر برای حذف الزامی است.');
                    if ($userIdToDelete === $CURRENT_USER_ID) throw new Exception('شما نمی‌توانید حساب کاربری خود را حذف کنید.');
                    $updatedUsers = array_filter($updatedUsers, fn($u) => ($u['id'] ?? null) !== $userIdToDelete);
                    $message = 'کاربر با موفقیت حذف شد.';
                    break;
                case 'save':
                     if (!$clientUser) throw new Exception('اطلاعات کاربر ارسال نشده است.');

/**
 * Helper function to check if a hex color is light or dark.
 *  - Added null/string check.
 */
function isLightColor($hexColor) {
    if (!$hexColor || !is_string($hexColor)) return false;
    $hexColor = ltrim($hexColor, '#');
    if (strlen($hexColor) == 3) { $r = hexdec(str_repeat(substr($hexColor, 0, 1), 2)); $g = hexdec(str_repeat(substr($hexColor, 1, 1), 2)); $b = hexdec(str_repeat(substr($hexColor, 2, 1), 2)); }
    elseif (strlen($hexColor) == 6) { $r = hexdec(substr($hexColor, 0, 2)); $g = hexdec(substr($hexColor, 2, 2)); $b = hexdec(substr($hexColor, 4, 2)); }
    else return false;
    $brightness = (($r * 299) + ($g * 587) + ($b * 114)) / 1000;
    return $brightness > 155; // Threshold
}
